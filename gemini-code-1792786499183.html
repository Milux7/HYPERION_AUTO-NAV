Arquitectura y Componentes del Robot

Detalles del hardware de control, comunicación inalámbrica WiFi y algoritmo de seguridad para esquivar obstáculos.

Red WiFi Propia (Access Point)

El microcontrolador ESP32 crea su propia red inalámbrica ("RobotESP32"). No requiere router externo ni conexión a Internet para operar.

Servidor Web HTTP Asíncrono

Aloja la interfaz gráfica Web-Joystick en su memoria Flash mediante `PROGMEM` y responde a peticiones HTTP GET inmediatas (/adelante, /parar, etc.).

Sistema Anticolisión

Un bucle continuo mide la distancia con el HC-SR04. Si detecta un objeto a menos de **15 cm**, anula el comando activo y detiene los motores de inmediato.

Driver de Motores L298N

Puente H doble que controla la dirección y polaridad de los dos motores DC. La velocidad PWM está regulada a un valor constante de 200 (sobre 255).

Sensor Ultrasónico HC-SR04

Ubicado en la parte frontal. Emite pulsos a 40kHz y calcula el tiempo de retorno del eco en microsegundos para obtener la distancia exacta.

Control Multiplataforma

Compatible con cualquier smartphone o computadora con navegador web. El joystick adapta su sensibilidad mediante APIs de eventos `Pointer` táctiles.

Mapeo Completo de Pines y Cableado

Distribución de conexiones de hardware entre la placa ESP32, el driver L298N y el sensor ultrasónico.

Código Fuente Oficial (Arduino C++ para ESP32)

Firmware completo con Servidor Web en memoria Flash y control de seguridad.

#include <WiFi.h>
#include <WebServer.h>

const char* ssid     = "RobotESP32";      
const char* password  = "robot1234";       

WebServer server(80);

const int PIN_TRIG = 5;
const int PIN_ECHO = 18;

const int IN1 = 26;
const int IN2 = 27;
const int ENA = 25;

const int IN3 = 14;
const int IN4 = 12;
const int ENB = 33;

const int DISTANCIA_MINIMA_CM = 15; 
String comandoActual = "parar";

void detener() {
  digitalWrite(IN1, LOW); digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW); digitalWrite(IN4, LOW);
}

void avanzar() {
  digitalWrite(IN1, HIGH); digitalWrite(IN2, LOW);
  digitalWrite(IN3, HIGH); digitalWrite(IN4, LOW);
}

void retroceder() {
  digitalWrite(IN1, LOW); digitalWrite(IN2, HIGH);
  digitalWrite(IN3, LOW); digitalWrite(IN4, HIGH);
}

void girarIzquierda() {
  digitalWrite(IN1, LOW); digitalWrite(IN2, HIGH);
  digitalWrite(IN3, HIGH); digitalWrite(IN4, LOW);
}

void girarDerecha() {
  digitalWrite(IN1, HIGH); digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW); digitalWrite(IN4, HIGH);
}

long leerDistanciaCM() {
  digitalWrite(PIN_TRIG, LOW);
  delayMicroseconds(2);
  digitalWrite(PIN_TRIG, HIGH);
  delayMicroseconds(10);
  digitalWrite(PIN_TRIG, LOW);

  long duracion = pulseIn(PIN_ECHO, HIGH, 30000); 
  if (duracion == 0) return 999; 
  return (duracion * 0.0343) / 2.0;
}

void setup() {
  Serial.begin(115200);

  pinMode(PIN_TRIG, OUTPUT);
  pinMode(PIN_ECHO, INPUT);
  pinMode(IN1, OUTPUT); pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT); pinMode(IN4, OUTPUT);
  pinMode(ENA, OUTPUT); pinMode(ENB, OUTPUT);

  analogWrite(ENA, 200);
  analogWrite(ENB, 200);

  detener();

  WiFi.softAP(ssid, password);
  Serial.print("Red creada: "); Serial.println(ssid);
  Serial.print("Servidor IP: "); Serial.println(WiFi.softAPIP());

  server.on("/", []() { server.send(200, "text/html", "HYPERION AUTO-NAV OK"); });
  server.on("/adelante", []() { comandoActual = "adelante"; server.send(200, "text/plain", "OK"); });
  server.on("/atras", []() { comandoActual = "atras"; server.send(200, "text/plain", "OK"); });
  server.on("/izquierda", []() { comandoActual = "izquierda"; server.send(200, "text/plain", "OK"); });
  server.on("/derecha", []() { comandoActual = "derecha"; server.send(200, "text/plain", "OK"); });
  server.on("/parar", []() { comandoActual = "parar"; server.send(200, "text/plain", "OK"); });
  server.on("/distancia", []() { server.send(200, "text/plain", String(leerDistanciaCM())); });

  server.begin();
}

void loop() {
  server.handleClient();
  long distancia = leerDistanciaCM();

  if (distancia <= DISTANCIA_MINIMA_CM && comandoActual == "adelante") {
    detener();
    return;
  }

  if (comandoActual == "adelante") avanzar();
  else if (comandoActual == "atras") retroceder();
  else if (comandoActual == "izquierda") girarIzquierda();
  else if (comandoActual == "derecha") girarDerecha();
  else detener();
}
